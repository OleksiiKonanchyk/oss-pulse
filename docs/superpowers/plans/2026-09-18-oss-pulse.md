# OSS Pulse Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a deployed, public dashboard that tracks the health of open-source repositories against the live GitHub REST API, as a portfolio piece demonstrating React/TypeScript SaaS front-end work.

**Architecture:** A client-only SPA with one hard boundary — `src/api/` knows HTTP and nothing about React, `src/features/` knows React and never calls `fetch`, and typed TanStack Query hooks sit between them. The interesting logic (Link-header pagination, error classification, the `202` statistics retry) therefore lives in pure functions that are tested without rendering anything.

**Tech Stack:** Vite, React, TypeScript (strict), TanStack Query, React Router v6, react-hook-form, zod, Tailwind v4, Recharts, Vitest, React Testing Library, MSW. Deployed to Vercel as static files.

**Spec:** `docs/superpowers/specs/2026-09-18-oss-pulse-design.md`

## Global Constraints

- **TypeScript strict mode on.** No `any` in committed code; use `unknown` and narrow.
- **No state-management library.** Server state is TanStack Query. The only client state is the tracked-repo list, which is a `localStorage`-backed hook.
- **Client-only.** No backend, no secrets in the repo. The optional GitHub token is stored in `localStorage` and sent only to `api.github.com`.
- **No write operations against the GitHub API.** Reads only.
- **Two chart types only:** a bar chart (commit activity, reused at small size as the overview sparkline) and a line chart (issue throughput).
- **Every error reaching a component is an `ApiError`** from the discriminated union in Task 3. No component renders a raw error string.
- **Accessibility is not a later pass:** keyboard navigation, visible focus rings, correct table semantics, `aria-live` on async state changes, and text alternatives for both charts.
- **Rate limit budget:** GitHub allows 60 requests/hour unauthenticated. Query `staleTime` is one hour and the cache persists to `localStorage`.
- **Commit after every task.** Conventional commit prefixes (`feat:`, `test:`, `chore:`).

```ts
export type PageLinks = {
  first?: string
  prev?: string
  next?: string
  last?: string
}

const RELATIONS = ['first', 'prev', 'next', 'last'] as const
type Relation = (typeof RELATIONS)[number]

function isRelation(value: string): value is Relation {
  return (RELATIONS as readonly string[]).includes(value)
}

/**
 * Parses GitHub's RFC 5988 `Link` header into the page URLs it names.
 *
 * Only the four pagination relations are kept; GitHub also sends documentation
 * relations on some endpoints, and those are noise for our purposes.
 */
export function parseLinkHeader(header: string | null): PageLinks {
  if (!header) return {}

  const links: PageLinks = {}

  for (const section of header.split(',')) {
    const match = section.match(/<([^>]+)>\s*;\s*rel="([^"]+)"/)
    if (!match) continue

    const [, url, rel] = match
    if (isRelation(rel)) links[rel] = url
  }

  return links
}
```

```ts
export type ApiError =
  | { kind: 'rate-limited'; resetAt: Date }
  | { kind: 'not-found' }
  | { kind: 'network' }
  | { kind: 'unknown'; status: number; message: string }

/** Carries a classified ApiError across the throw boundary into TanStack Query. */
export class GitHubError extends Error {
  readonly error: ApiError

  constructor(error: ApiError) {
    super(error.kind)
    this.name = 'GitHubError'
    this.error = error
  }
}

export function isApiError(value: unknown): value is GitHubError {
  return value instanceof GitHubError
}

const ONE_HOUR_MS = 3_600_000

function messageFrom(body: unknown, fallback: string): string {
  if (body && typeof body === 'object' && 'message' in body) {
    const { message } = body as { message: unknown }
    if (typeof message === 'string' && message.length > 0) return message
  }
  return fallback
}

/**
 * A 403 from GitHub means two very different things depending on the rate-limit
 * headers: quota exhausted (recoverable, with a known reset time) or genuinely
 * forbidden. Only the first earns the countdown screen.
 */
export function classifyResponse(response: Response, body: unknown): ApiError {
  const remaining = response.headers.get('x-ratelimit-remaining')
  const exhausted = remaining === '0'

  if ((response.status === 403 || response.status === 429) && exhausted) {
    const reset = response.headers.get('x-ratelimit-reset')
    const resetAt = reset
      ? new Date(Number(reset) * 1000)
      : new Date(Date.now() + ONE_HOUR_MS)

    return { kind: 'rate-limited', resetAt }
  }

  if (response.status === 404) return { kind: 'not-found' }

  return {
    kind: 'unknown',
    status: response.status,
    message: messageFrom(body, response.statusText || `HTTP ${response.status}`),
  }
}
```

Create `src/api/rate-limit.ts`:

```ts
export type RateLimit = {
  limit: number
  remaining: number
  resetAt: Date
}

export function readRateLimit(headers: Headers): RateLimit | null {
  const limit = headers.get('x-ratelimit-limit')
  const remaining = headers.get('x-ratelimit-remaining')
  const reset = headers.get('x-ratelimit-reset')

  if (limit === null || remaining === null || reset === null) return null

  return {
    limit: Number(limit),
    remaining: Number(remaining),
    resetAt: new Date(Number(reset) * 1000),
  }
}

type Listener = (limit: RateLimit) => void

const listeners = new Set<Listener>()

/**
 * Every response carries the current quota, so the header indicator is fed by a
 * side channel rather than by a dedicated request that would itself cost quota.
 */
export function subscribeToRateLimit(listener: Listener): () => void {
  listeners.add(listener)
  return () => listeners.delete(listener)
}

export function publishRateLimit(limit: RateLimit): void {
  for (const listener of listeners) listener(limit)
}
```

- [ ] **Step 5: Implement the client**

Create `src/api/client.ts`:

```ts
import { classifyResponse, GitHubError } from './errors'
import { parseLinkHeader, type PageLinks } from './link-header'
import { publishRateLimit, readRateLimit, type RateLimit } from './rate-limit'

const BASE_URL = 'https://api.github.com'

export type GitHubResponse<T> = {
  data: T
  links: PageLinks
  rateLimit: RateLimit | null
}

export type FetchOptions = {
  token?: string | null
  signal?: AbortSignal
}

async function readBody(response: Response): Promise<unknown> {
  const text = await response.text()
  if (text.length === 0) return null

  try {
    return JSON.parse(text) as unknown
  } catch {
    return null
  }
}

/**
 * The single point where this app talks HTTP. Absolute URLs are accepted so that
 * a `Link` header URL can be followed directly without re-deriving its query.
 */
export async function githubFetch<T>(
  path: string,
  { token, signal }: FetchOptions = {},
): Promise<GitHubResponse<T>> {
  const url = path.startsWith('http') ? path : `${BASE_URL}${path}`

  const headers: Record<string, string> = {
    Accept: 'application/vnd.github+json',
    'X-GitHub-Api-Version': '2022-11-28',
  }
  if (token) headers.Authorization = `Bearer ${token}`

  let response: Response
  try {
    response = await fetch(url, { headers, signal })
  } catch (cause) {
    if (cause instanceof DOMException && cause.name === 'AbortError') throw cause
    throw new GitHubError({ kind: 'network' })
  }

  const rateLimit = readRateLimit(response.headers)
  if (rateLimit) publishRateLimit(rateLimit)

  const body = await readBody(response)

  if (!response.ok) throw new GitHubError(classifyResponse(response, body))

  return {
    data: body as T,
    links: parseLinkHeader(response.headers.get('link')),
    rateLimit,
  }
}
```

```ts
import { githubFetch, type FetchOptions } from './client'
import type { CommitWeek, Contributor, Issue, IssueFilters, Repo } from './types'

type RepoPayload = {
  full_name: string
  description: string | null
  stargazers_count: number
  forks_count: number
  open_issues_count: number
  language: string | null
  pushed_at: string
  html_url: string
  license: { spdx_id: string } | null
}

export async function getRepo(fullName: string, options?: FetchOptions): Promise<Repo> {
  const { data } = await githubFetch<RepoPayload>(`/repos/${fullName}`, options)

  return {
    fullName: data.full_name,
    description: data.description,
    stars: data.stargazers_count,
    forks: data.forks_count,
    openIssues: data.open_issues_count,
    language: data.language,
    pushedAt: new Date(data.pushed_at),
    htmlUrl: data.html_url,
    license: data.license?.spdx_id ?? null,
  }
}

type ContributorPayload = {
  login: string
  avatar_url: string
  contributions: number
  html_url: string
}

export async function getContributors(
  fullName: string,
  options?: FetchOptions,
): Promise<Contributor[]> {
  const { data } = await githubFetch<ContributorPayload[]>(
    `/repos/${fullName}/contributors?per_page=10`,
    options,
  )

  return data.map((entry) => ({
    login: entry.login,
    avatarUrl: entry.avatar_url,
    contributions: entry.contributions,
    htmlUrl: entry.html_url,
  }))
}

type IssuePayload = {
  number: number
  title: string
  state: 'open' | 'closed'
  user: { login: string } | null
  labels: { name: string; color: string }[]
  comments: number
  created_at: string
  html_url: string
  pull_request?: unknown
}

/**
 * `/issues` returns pull requests too — GitHub models them as issues. Every
 * consumer here means issues in the colloquial sense, so they are filtered out
 * at the boundary rather than in each caller.
 */
export async function getIssues(
  fullName: string,
  filters: IssueFilters,
  options?: FetchOptions,
): Promise<{ issues: Issue[]; hasNextPage: boolean }> {
  const params = new URLSearchParams({
    state: filters.state,
    page: String(filters.page),
    per_page: '25',
  })
  if (filters.labels) params.set('labels', filters.labels)
  if (filters.creator) params.set('creator', filters.creator)

  const { data, links } = await githubFetch<IssuePayload[]>(
    `/repos/${fullName}/issues?${params}`,
    options,
  )

  return {
    issues: data
      .filter((entry) => entry.pull_request === undefined)
      .map((entry) => ({
        number: entry.number,
        title: entry.title,
        state: entry.state,
        author: entry.user?.login ?? null,
        labels: entry.labels,
        comments: entry.comments,
        createdAt: new Date(entry.created_at),
        htmlUrl: entry.html_url,
      })),
    hasNextPage: links.next !== undefined,
  }
}

type CommitActivityPayload = { week: number; total: number; days: number[] }

const DEFAULT_ATTEMPTS = 5
const DEFAULT_DELAY_MS = 1500

function wait(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms))
}

/**
 * GitHub computes repository statistics asynchronously and answers `202` with an
 * empty body while it works. Returning 'computing' rather than throwing lets the
 * UI say what is actually happening instead of showing a failure.
 */
export async function getCommitActivity(
  fullName: string,
  options: FetchOptions & { attempts?: number; delayMs?: number } = {},
): Promise<CommitWeek[] | 'computing'> {
  const { attempts = DEFAULT_ATTEMPTS, delayMs = DEFAULT_DELAY_MS, ...fetchOptions } = options

  for (let attempt = 0; attempt < attempts; attempt += 1) {
    const { data } = await githubFetch<CommitActivityPayload[] | null>(
      `/repos/${fullName}/stats/commit_activity`,
      fetchOptions,
    )

    if (Array.isArray(data)) {
      return data.map((week) => ({
        weekStart: new Date(week.week * 1000),
        total: week.total,
      }))
    }

    if (attempt < attempts - 1) await wait(delayMs)
  }

  return 'computing'
}
```

```ts
/**
 * localStorage throws in private-browsing modes and can hold anything a previous
 * version wrote, so every read is defensive and every failure is non-fatal.
 */
export function readJson<T>(key: string, fallback: T): T {
  try {
    const raw = localStorage.getItem(key)
    if (raw === null) return fallback
    return JSON.parse(raw) as T
  } catch {
    return fallback
  }
}

export function writeJson(key: string, value: unknown): void {
  try {
    localStorage.setItem(key, JSON.stringify(value))
  } catch {
    // Storage unavailable or full: the app still works for this session.
  }
}
```

```ts
import { readJson, writeJson } from '../../lib/storage'

const KEY = 'oss-pulse:tracked-repos'

/** Recognisable, high-activity repositories so a first visit is never blank. */
export const DEFAULT_TRACKED_REPOS = ['facebook/react', 'vercel/next.js', 'rails/rails']

type Listener = () => void
const listeners = new Set<Listener>()

function isStringArray(value: unknown): value is string[] {
  return Array.isArray(value) && value.every((entry) => typeof entry === 'string')
}

export function getTrackedRepos(): string[] {
  const stored = readJson<unknown>(KEY, null)
  if (stored === null) return DEFAULT_TRACKED_REPOS
  if (!isStringArray(stored)) return DEFAULT_TRACKED_REPOS

  return stored
}

function save(repos: string[]): void {
  writeJson(KEY, repos)
  for (const listener of listeners) listener()
}

export function trackRepo(fullName: string): void {
  const repos = getTrackedRepos()
  const exists = repos.some((name) => name.toLowerCase() === fullName.toLowerCase())
  if (exists) return

  save([...repos, fullName])
}

export function untrackRepo(fullName: string): void {
  const repos = getTrackedRepos().filter(
    (name) => name.toLowerCase() !== fullName.toLowerCase(),
  )

  save(repos)
}

export function subscribeToTrackedRepos(listener: Listener): () => void {
  listeners.add(listener)
  return () => listeners.delete(listener)
}
```

```ts
export type Theme = 'light' | 'dark' | 'system'

type Listener = () => void
const listeners = new Set<Listener>()

export function getTheme(): Theme {
  const stored = readJson<unknown>(KEY, 'system')
  return stored === 'light' || stored === 'dark' ? stored : 'system'
}

export function setTheme(theme: Theme): void {
  writeJson(KEY, theme)
  applyTheme(theme)
  for (const listener of listeners) listener()
}

export function applyTheme(theme: Theme): void {
  const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches
  const dark = theme === 'dark' || (theme === 'system' && prefersDark)

  document.documentElement.classList.toggle('dark', dark)
}

function subscribe(listener: Listener): () => void {
  listeners.add(listener)
  return () => listeners.delete(listener)
}

export function useTheme(): Theme {
  return useSyncExternalStore(subscribe, getTheme)
```

```ts
export function formatCompact(value: number): string {
  if (value < 1000) return String(value)
  if (value < 1_000_000) return `${(value / 1000).toFixed(1).replace(/\.0$/, '')}k`

  return `${(value / 1_000_000).toFixed(1).replace(/\.0$/, '')}M`
}

export function formatRelativeDate(date: Date): string {
  const days = Math.round((Date.now() - date.getTime()) / 86_400_000)

  if (days < 1) return 'today'
  if (days === 1) return 'yesterday'
  if (days < 30) return `${days} days ago`
  if (days < 365) return `${Math.round(days / 30)} months ago`

  return `${Math.round(days / 365)} years ago`
}
```

```tsx
import { useState } from 'react'
import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import { isApiError } from '../../api/errors'
import { verifyRepoExists } from '../../queries/use-github-queries'
import { useTrackedRepos } from './use-tracked-repos'

const OWNER_NAME = /^([\w.-]+)\/([\w.-]+)$/

/** Accepts `owner/name` or any github.com URL pointing at a repository. */
export function parseRepoInput(value: string): string | null {
  const trimmed = value.trim()
  if (trimmed.length === 0) return null

  if (trimmed.startsWith('http')) {
    try {
      const url = new URL(trimmed)
      if (url.hostname !== 'github.com') return null

      const [owner, name] = url.pathname.split('/').filter(Boolean)
      if (!owner || !name) return null

      return `${owner}/${name}`
    } catch {
      return null
    }
  }

  return OWNER_NAME.test(trimmed) ? trimmed : null
}

const schema = z.object({
  input: z.string().refine((value) => parseRepoInput(value) !== null, {
    message: 'Use owner/name, or paste a GitHub repository URL.',
  }),
})

type FormValues = z.infer<typeof schema>

export function AddRepoForm() {
  const { track, untrack } = useTrackedRepos()
  const [serverError, setServerError] = useState<string | null>(null)

  const { register, handleSubmit, reset, formState } = useForm<FormValues>({
    resolver: zodResolver(schema),
    defaultValues: { input: '' },
  })

  async function onSubmit({ input }: FormValues) {
    const fullName = parseRepoInput(input)
    if (!fullName) return

    setServerError(null)

    // Optimistic: the row appears immediately and rolls back if GitHub disagrees.
    track(fullName)
    reset()

    try {
      await verifyRepoExists(fullName)
    } catch (error) {
      untrack(fullName)
      setServerError(
        isApiError(error) && error.error.kind === 'not-found'
          ? `${fullName} could not be found on GitHub.`
          : `${fullName} could not be verified. Try again.`,
      )
    }
  }
```

```tsx
import { Line, LineChart, ResponsiveContainer, Tooltip, XAxis, YAxis } from 'recharts'
import type { Issue } from '../../api/types'

function weekKey(date: Date): string {
  const monday = new Date(date)
  monday.setUTCDate(monday.getUTCDate() - ((monday.getUTCDay() + 6) % 7))

  return monday.toISOString().slice(0, 10)
}

export function IssueThroughputChart({ issues }: { issues: Issue[] }) {
  if (issues.length === 0) {
    return <p className="text-sm text-slate-500">No issues to chart.</p>
  }

  const counts = new Map<string, number>()
  for (const issue of issues) {
    const key = weekKey(issue.createdAt)
    counts.set(key, (counts.get(key) ?? 0) + 1)
  }

  const data = [...counts.entries()]
    .sort(([a], [b]) => a.localeCompare(b))
    .map(([label, opened]) => ({ label, opened }))

  return (
    <figure
      role="img"
      aria-label={`Issues opened per week: ${issues.length} issues across ${data.length} weeks.`}
    >
      <ResponsiveContainer width="100%" height={220}>
        <LineChart data={data}>
          <XAxis dataKey="label" tick={{ fontSize: 12 }} />
          <YAxis width={32} tick={{ fontSize: 12 }} />
          <Tooltip />
          <Line
            type="monotone"
            dataKey="opened"
            stroke="currentColor"
            className="text-violet-600 dark:text-violet-400"
            dot={false}
          />
        </LineChart>
      </ResponsiveContainer>
    </figure>
  )
}
```

Create `src/features/issues/use-issues-params.ts`:

```ts
import { useCallback } from 'react'
import { useSearchParams } from 'react-router-dom'
import type { IssueFilters } from '../../api/types'

const STATES = ['open', 'closed', 'all'] as const

function readState(value: string | null): IssueFilters['state'] {
  return (STATES as readonly string[]).includes(value ?? '')
    ? (value as IssueFilters['state'])
    : 'open'
}

function readPage(value: string | null): number {
  const page = Number(value)
  return Number.isInteger(page) && page > 0 ? page : 1
}

export function useIssuesParams() {
  const [params, setParams] = useSearchParams()

  const filters: IssueFilters = {
    state: readState(params.get('state')),
    labels: params.get('labels') ?? undefined,
    creator: params.get('creator') ?? undefined,
    page: readPage(params.get('page')),
  }

  /**
   * Any change other than paging resets to page one — page 7 of a different
   * filter is almost always empty, and landing there reads as a bug.
   */
  const setFilters = useCallback(
    (next: Partial<IssueFilters>) => {
      const merged = { ...filters, ...next }
      const pagingOnly = Object.keys(next).length === 1 && 'page' in next
      const updated = new URLSearchParams()

      if (merged.state !== 'open') updated.set('state', merged.state)
      if (merged.labels) updated.set('labels', merged.labels)
      if (merged.creator) updated.set('creator', merged.creator)

      const page = pagingOnly ? merged.page : 1
      if (page > 1) updated.set('page', String(page))

      setParams(updated)
    },
    [filters, setParams],
  )

  return { filters, setFilters }
}
```
